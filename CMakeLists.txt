cmake_minimum_required(VERSION 3.24)
project(pcap_c17 VERSION 2.0.0 LANGUAGES C CUDA)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
 set(CMAKE_CUDA_ARCHITECTURES 89)
endif()
find_package(CUDAToolkit REQUIRED)
include(FetchContent)
FetchContent_Declare(optix_dev GIT_REPOSITORY https://github.com/NVIDIA/optix-dev.git GIT_TAG v9.1.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(optix_dev)
set(OPTIX_INCLUDE_DIR ${optix_dev_SOURCE_DIR}/include)
set(DEVICE_IR ${CMAKE_CURRENT_BINARY_DIR}/device.optixir)
set(DEVICE_EMBED ${CMAKE_CURRENT_BINARY_DIR}/device_embed.c)
add_custom_command(OUTPUT ${DEVICE_IR} COMMAND ${CMAKE_CUDA_COMPILER} -optix-ir -std=c++14 -arch=compute_75 -Wno-deprecated-gpu-targets -I${OPTIX_INCLUDE_DIR} -I${CMAKE_CURRENT_SOURCE_DIR}/src_c ${CMAKE_CURRENT_SOURCE_DIR}/src_c/device.cu -o ${DEVICE_IR} DEPENDS src_c/device.cu src_c/rt_params.h src_c/common.h VERBATIM)
add_custom_command(OUTPUT ${DEVICE_EMBED} COMMAND ${CMAKE_COMMAND} -DINPUT=${DEVICE_IR} -DOUTPUT=${DEVICE_EMBED} -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/EmbedC.cmake DEPENDS ${DEVICE_IR} cmake/EmbedC.cmake VERBATIM)
add_library(pcap_core STATIC src_c/common.c src_c/search.c src_c/planner.c src_c/rt_backend.c src_c/grid_backend.c src_c/grid_kernels.cu ${DEVICE_EMBED})
target_include_directories(pcap_core PUBLIC src_c ${OPTIX_INCLUDE_DIR} ${CUDAToolkit_INCLUDE_DIRS})
target_link_libraries(pcap_core PUBLIC CUDA::cudart CUDA::cuda_driver)
if(MSVC)
 target_compile_definitions(pcap_core PUBLIC _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
 target_compile_options(pcap_core PRIVATE $<$<COMPILE_LANGUAGE:C>:/W4> $<$<COMPILE_LANGUAGE:C>:/fp:strict>)
else()
 target_link_libraries(pcap_core PUBLIC m dl)
 target_compile_options(pcap_core PRIVATE $<$<COMPILE_LANGUAGE:C>:-Wall> $<$<COMPILE_LANGUAGE:C>:-ffp-contract=off>)
endif()
add_executable(ooc_rtnn src_c/main.c)
target_link_libraries(ooc_rtnn PRIVATE pcap_core)
add_executable(radius_demo examples/radius_demo.c)
target_link_libraries(radius_demo PRIVATE pcap_core)
if(MSVC)
 target_compile_options(radius_demo PRIVATE /W4 /fp:strict)
else()
 target_compile_options(radius_demo PRIVATE -Wall -ffp-contract=off)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(c_core_tests tests/test_c_core.c src_c/common.c)
 target_include_directories(c_core_tests PRIVATE src_c)
 if(MSVC)
 target_compile_definitions(c_core_tests PRIVATE _CRT_SECURE_NO_WARNINGS)
 target_compile_options(c_core_tests PRIVATE /fp:strict)
 else()
 target_link_libraries(c_core_tests PRIVATE m)
 endif()
 add_test(NAME c_core COMMAND c_core_tests)
endif()
